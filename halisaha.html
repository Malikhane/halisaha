<script>

let data = JSON.parse(localStorage.getItem("halisaha")) || [];


function submitForm() {

  const name = document.getElementById("name").value.trim();

  const status = document.getElementById("status").value;

  const reason = document.getElementById("reason").value;


  if (!name) return alert("İsim gir!");


  let team = null;

  if (status === "geliyor") {

    team = Math.random() < 0.5 ? "A" : "B";

  }


  data.push({ name, status, reason, team });

  localStorage.setItem("halisaha", JSON.stringify(data));


  render();

}


function render() {

  const list = document.getElementById("participants");

  const teamA = document.getElementById("teamA");

  const teamB = document.getElementById("teamB");


  list.innerHTML = "";

  teamA.innerHTML = "";

  teamB.innerHTML = "";


  data.forEach(p => {

    const li = document.createElement("li");

    li.textContent = p.name + " - " + p.status +

      (p.status === "gelmiyor" ? " (" + p.reason + ")" : "");

    list.appendChild(li);


    if (p.team === "A") {

      teamA.innerHTML += `<div class="player">${p.name}</div>`;

    }

    if (p.team === "B") {

      teamB.innerHTML += `<div class="player">${p.name}</div>`;

    }

  });

}


render();

</script>
